CUDA_VISIBLE_DEVICES=0 nohup bash -ec 'python -c "import torch; assert torch.cuda.is_available(), \"CUDA unavailable: training stopped\"; torch.ones(1, device=\"cuda\"); print(torch.cuda.get_device_name(0))"; exec python -u launch.py algorithms/two_stage_gat/train.py --task S --resource S --deadline Tight --episodes 600 --model-seed 1 --experiment-protocol configs/f22_split_v1.json --normalization-mode f19_shared_v1 --device cuda' > outputs/server_console_logs/two_stage_gat_SS_Tight_ep600_seed1.log 2>&1 & echo $! > outputs/server_console_logs/two_stage_gat_SS_Tight_ep600_seed1.pid
