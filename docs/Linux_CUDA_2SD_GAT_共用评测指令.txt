# 【2SD-GAT-Adapted（两阶段GAT-PPO）：two_stage_gat】
# --------------------------------------------------------------
# Linux / Bash；在项目根目录、已激活的 Python 环境中执行。
# 先完成对应源训练，再启动评测；每组依次评测 S、M、L。
# 源训练始终 task=S，资源/截止期不变，加载对应 best.pth。

# 2SD-GAT-Adapted（两阶段GAT-PPO） | 源训练SS | Tight | SS -> MS -> LS
mkdir -p outputs/server_eval_logs
CUDA_VISIBLE_DEVICES=0 nohup bash -ec 'python -c "import torch; assert torch.cuda.is_available(), \"CUDA unavailable: evaluation stopped\"; torch.ones(1, device=\"cuda\"); print(torch.cuda.get_device_name(0))"; for task in S M L; do echo "=== two_stage_gat task=${task} resource=S deadline=Tight seeds=700-729 source=SS ==="; python -u launch.py algorithms/two_stage_gat/evaluate.py --task "$task" --resource S --deadline Tight --seeds 700-729 --train-task S --model-seed 1 --experiment-protocol configs/f22_split_v1.json --normalization-mode f19_shared_v1 --device cuda; done' > outputs/server_eval_logs/eval_two_stage_gat_sourceSS_Tight_model1.log 2>&1 &
echo $! > outputs/server_eval_logs/eval_two_stage_gat_sourceSS_Tight_model1.pid
