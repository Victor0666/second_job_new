# 【DRL-GNN（公共状态输入适配版）：drl_gnn】
# --------------------------------------------------------------
# Linux / Bash；在项目根目录、已激活的 Python 环境中执行。
# 输入版本：public_raw_task8_vm9_bn_v2（Task8 / VM9）；需要重新训练，旧 Task4 / VM3 权重不兼容。
# 调度v3：父任务已排程即释放子任务，允许全部兼容VM的队尾未来预留；旧 Ready×idle Task8/VM9 模型也需重训。
# CUDA_VISIBLE_DEVICES=0；每个命令是一项独立后台任务，按需单独执行。
# --normalization-mode f19_shared_v1 保留公共参考元数据；策略仍使用 raw 特征和单图 BN。
# PPO、奖励和最低验证能耗选模不变；本次未修改已有的大图 minibatch 显存处理。
# 评测：先完成对应源模型训练，加载新版 best.pth，测试种子 700-729。
# 每个源模型依次评测 task=S/M/L；资源和截止期保持对应源训练设置，共 27 个场景。

# DRL-GNN（Task8 / VM9） | 源训练SS | Tight | SS -> MS -> LS
mkdir -p outputs/server_eval_logs
CUDA_VISIBLE_DEVICES=0 nohup bash -ec 'python -c "import torch; assert torch.cuda.is_available(), \"CUDA unavailable: stopped\"; torch.ones(1, device=\"cuda\"); print(torch.cuda.get_device_name(0))"; for task in S M L; do echo "=== drl_gnn task=${task} resource=S deadline=Tight seeds=700-729 source=SS input=public_raw_task8_vm9_bn_v2 ==="; python -u launch.py algorithms/drl_gnn/evaluate.py --task "$task" --resource S --deadline Tight --seeds 700-729 --train-task S --model-seed 1 --experiment-protocol configs/f22_split_v1.json --normalization-mode f19_shared_v1 --device cuda; done' > outputs/server_eval_logs/eval_drl_gnn_sourceSS_Tight_task8_vm9_parent_scheduled_v3_model1.log 2>&1 &
echo $! > outputs/server_eval_logs/eval_drl_gnn_sourceSS_Tight_task8_vm9_parent_scheduled_v3_model1.pid
