# 在仓库根目录执行。每次调用启动一个后台训练进程；按服务器容量分批选用调用。
# IRWS-R-FE：保留V3加速机制，验证总体工作流成功率>=99%后选最低能耗。
# 固定配置：600回合、模型种子1、q10、dt5、rollout64、float32、F22/F19。
mkdir -p outputs/server_console_logs

train_irws_r_fe() {
    local res="$1" ddl="$2"
    local prefix="outputs/server_console_logs/irws_r_fe_S${res}_${ddl}_v3_fe_v1_select_sr099_minenergy_v1_ep600_model1_q10_dt5_n64_float32"

    CUDA_VISIBLE_DEVICES=0 nohup python -u launch.py algorithms/irws_r_fe/train.py \
        --task S --resource "$res" --deadline "$ddl" \
        --episodes 600 --model-seed 1 \
        --experiment-protocol configs/f22_split_v1.json \
        --normalization-mode f19_shared_v1 \
        --queue-capacity 10 --slot-duration 5 --rollout-size 64 --precision float32 \
        --device cuda \
        > "${prefix}.log" 2>&1 &
    echo $! > "${prefix}.pid"
}

# SS：小任务、小资源
train_irws_r_fe S Tight
train_irws_r_fe S Medium
train_irws_r_fe S Loose

# SM：小任务、中资源
train_irws_r_fe M Tight
train_irws_r_fe M Medium
train_irws_r_fe M Loose

# SL：小任务、大资源
train_irws_r_fe L Tight
train_irws_r_fe L Medium
train_irws_r_fe L Loose
