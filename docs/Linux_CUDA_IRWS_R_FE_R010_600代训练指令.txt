# Linux / Bash: run from the project root with matching F22 cache and F19 calibration.
# R010 delay/energy weights: 0.1/0.9. One invocation starts one background process.
# Run the nine source scenes in batches according to GPU capacity.
train_irws_r_fe_r010() {
    local res="$1" ddl="$2"
    local prefix="outputs/server_console_logs/train_irws_r_fe_r010_S${res}_${ddl}_m1"

    mkdir -p outputs/server_console_logs

    CUDA_VISIBLE_DEVICES=0 nohup python -u launch.py \
        algorithms/irws_r_fe_r010/train.py \
        --task S --resource "$res" --deadline "$ddl" \
        --episodes 600 --model-seed 1 \
        --experiment-protocol configs/f22_split_v1.json \
        --normalization-mode f19_shared_v1 --device cuda \
        --queue-capacity 10 --slot-duration 5 --rollout-size 64 --precision float32 \
        > "${prefix}.log" 2>&1 &

    echo $! > "${prefix}.pid"
}

# SS source
train_irws_r_fe_r010 S Tight
train_irws_r_fe_r010 S Medium
train_irws_r_fe_r010 S Loose

# SM source
train_irws_r_fe_r010 M Tight
train_irws_r_fe_r010 M Medium
train_irws_r_fe_r010 M Loose

# SL source
train_irws_r_fe_r010 L Tight
train_irws_r_fe_r010 L Medium
train_irws_r_fe_r010 L Loose
