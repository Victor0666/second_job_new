# 方案A：只分配空闲VM，保留5秒调度tick；q10仅兼容观测布局，不启用VM等待队列。
# 本变体固定超期奖励权重0.9、能耗权重0.1，原FE的0.75版本保留。
# 在仓库根目录执行。每次调用启动一个后台训练进程；按服务器容量分批选用调用。
# IRWS-R-FE_R090_IDLE：保留V3加速机制，验证总体工作流成功率>=99%后选最低能耗。
# 固定配置：600回合、模型种子1、q10、dt5、rollout64、float32、F22/F19。
train_irws_r_fe_r090_idle() {
    local res="$1" ddl="$2"
    local prefix="outputs/server_train_logs/train_irws_r_fe_r090_idle_S${res}_${ddl}_m1_ep600"

    mkdir -p outputs/server_train_logs

    CUDA_VISIBLE_DEVICES=0 nohup bash -c '
        res="$1"
        ddl="$2"

        exec python -u launch.py algorithms/irws_r_fe_r090_idle/train.py \
            --task S --resource "$res" --deadline "$ddl" \
            --episodes 600 --model-seed 1 \
            --experiment-protocol configs/f22_split_v1.json \
            --normalization-mode f19_shared_v1 \
            --queue-capacity 10 --slot-duration 5 --rollout-size 64 --precision float32 \
            --device cuda
    ' irws_r_fe_r090_idle_train "$res" "$ddl" > "${prefix}.log" 2>&1 &

    echo $! > "${prefix}.pid"
}

# SS：小任务、小资源
train_irws_r_fe_r090_idle S Tight
train_irws_r_fe_r090_idle S Medium
train_irws_r_fe_r090_idle S Loose

# SM：小任务、中资源
train_irws_r_fe_r090_idle M Tight
train_irws_r_fe_r090_idle M Medium
train_irws_r_fe_r090_idle M Loose

# SL：小任务、大资源
train_irws_r_fe_r090_idle L Tight
train_irws_r_fe_r090_idle L Medium
train_irws_r_fe_r090_idle L Loose
