# Linux / Bash：MAHDRL-GE、MAHDRL-PF 共用评测指令（适用于50代或600代训练后的模型）
# 从项目根目录、已激活的 Python/CUDA 环境执行；每组为可单独复制的后台命令。
# 每个算法：9个源训练场景 × 每组3个目标场景 × 30个实例 = 810次实例评测；两算法合计1620次。
# resource=S：SS -> MS -> LS；resource=M：SM -> MM -> LM；resource=L：SL -> ML -> LL。
# 同组固定资源、截止期，始终加载 --train-task S 对应的本算法 best 三层模型。
# 评测指令不区分训练代数：加载当前默认模型目录内的best，不一定是最后一代模型。
# 50/600代的默认评测CSV、统一测量目录及这里的评测.log/.pid名称相同，保存两批结果须在下一批前归档。
# 同一源训练场景的训练完成后再执行其评测，避免评测时模型文件被训练任务更新。
# 700-729沿用现行F22评测名单；不用于反复挑选修改方案。

# 【MAHDRL-GE（引导探索）：mahdrl_ge】
# --------------------------------------------------------------

# MAHDRL-GE（引导探索） | 源训练SS | Tight | SS -> MS -> LS
mkdir -p outputs/server_eval_logs
CUDA_VISIBLE_DEVICES=0 nohup bash -ec 'python -c '"'"'import torch; assert torch.cuda.is_available(), "CUDA unavailable: evaluation stopped"; torch.ones(1, device="cuda"); print(torch.cuda.get_device_name(0))'"'"'; for task in S M L; do echo "=== mahdrl_ge task=${task} resource=S deadline=Tight seeds=700-729 source=SS ==="; python -u launch.py algorithms/mahdrl_ge/evaluate.py --task "$task" --resource S --deadline Tight --seeds 700-729 --model-seed 1 --experiment-protocol configs/f22_split_v1.json --train-task S --normalization-mode f19_shared_v1; done' > outputs/server_eval_logs/eval_mahdrl_ge_sourceSS_Tight_model1.log 2>&1 &
echo $! > outputs/server_eval_logs/eval_mahdrl_ge_sourceSS_Tight_model1.pid
