# 在服务器项目根目录执行。默认：600回合、task S、model seed 1、VA预评估开启、HA周期4、GPU0。
# 先粘贴函数定义，再按需要运行下面九组训练命令。

train_mahdrl_as_fast_hscore() {
    local res="$1" ddl="$2"
    local preview="${3:-on}" period="${4:-4}" seed="${5:-1}"
    local preview_flag="--va-preview" p=1
    case "$preview" in
        on) ;;
        off) preview_flag="--no-va-preview"; p=0 ;;
        *) printf 'preview must be on/off\n' >&2; return 2 ;;
    esac
    local prefix="outputs/server_train_logs/train_hscore_S${res}_${ddl}_p${p}h${period}_sr092_m${seed}_$(date +%Y%m%d_%H%M%S)_${RANDOM}"
    mkdir -p outputs/server_train_logs

    CUDA_VISIBLE_DEVICES=${GPU:-0} nohup bash -c '
        res="$1"
        ddl="$2"
        preview_flag="$3"
        period="$4"
        seed="$5"

        echo "[START] train=S${res}/${ddl} preview=${preview_flag} HA-period=${period} model=${seed}"
        python -u launch.py \
            algorithms/mahdrl_as_fast_hscore/train.py \
            --task S --resource "$res" --deadline "$ddl" \
            --episodes 600 --model-seed "$seed" \
            --experiment-protocol configs/f22_split_v1.json \
            --normalization-mode f19_shared_v1 --device cuda \
            "$preview_flag" --ha-update-period "$period" \
            --diagnostics-every 100 \
            --profile-mode sampled --profile-every 100 \
            || exit 1
        echo "[DONE] train=S${res}/${ddl} model=${seed}"
    ' mahdrl_as_fast_hscore_train "$res" "$ddl" "$preview_flag" "$period" "$seed" \
        > "${prefix}.log" 2>&1 < /dev/null &

    echo $! > "${prefix}.pid"
    printf 'started: %s.log\n' "$prefix"
}

train_mahdrl_as_fast_hscore S Tight
train_mahdrl_as_fast_hscore S Medium
train_mahdrl_as_fast_hscore S Loose
train_mahdrl_as_fast_hscore M Tight
train_mahdrl_as_fast_hscore M Medium
train_mahdrl_as_fast_hscore M Loose
train_mahdrl_as_fast_hscore L Tight
train_mahdrl_as_fast_hscore L Medium
train_mahdrl_as_fast_hscore L Loose
