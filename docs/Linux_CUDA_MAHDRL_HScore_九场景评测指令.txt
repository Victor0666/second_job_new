# 在服务器项目根目录执行。默认：seeds700-729、model seed1、VA预评估开启、HA周期4、GPU0。
# 先粘贴函数定义，再按需要运行下面九组评测命令。

eval_mahdrl_as_fast_hscore_all() {
    local res="$1" ddl="$2"
    local targets="${3:-S M L}"
    local preview="${4:-on}" period="${5:-4}" seed="${6:-1}"
    local preview_flag="--va-preview" p=1
    case "$preview" in
        on) ;;
        off) preview_flag="--no-va-preview"; p=0 ;;
        *) printf 'preview must be on/off\n' >&2; return 2 ;;
    esac
    local prefix="outputs/server_eval_logs/eval_hscore_S${res}_${ddl}_targets${targets// /}_p${p}h${period}_sr092_s700-729_m${seed}_$(date +%Y%m%d_%H%M%S)_${RANDOM}"
    mkdir -p outputs/server_eval_logs

    CUDA_VISIBLE_DEVICES=${GPU:-0} nohup bash -c '
        res="$1"
        ddl="$2"
        targets="$3"
        preview_flag="$4"
        period="$5"
        seed="$6"

        for task in $targets; do
            echo "[START] source=S${res}/${ddl} target=${task}${res}/${ddl}"

            python -u launch.py \
                algorithms/mahdrl_as_fast_hscore/evaluate.py \
                --task "$task" --resource "$res" --deadline "$ddl" \
                --train-task S --train-resource "$res" \
                --train-deadline "$ddl" \
                --seeds 700-729 --model-seed "$seed" \
                --experiment-protocol configs/f22_split_v1.json \
                --normalization-mode f19_shared_v1 --device cuda \
                "$preview_flag" --ha-update-period "$period" \
                --diagnostics-every 100 \
                --profile-mode sampled --profile-every 100 \
                || exit 1

            echo "[DONE] source=S${res}/${ddl} target=${task}${res}/${ddl}"
        done

        echo "[ALL_DONE] source=S${res}/${ddl} targets=${targets}"
    ' mahdrl_as_fast_hscore_eval "$res" "$ddl" "$targets" "$preview_flag" "$period" "$seed" \
        > "${prefix}.log" 2>&1 < /dev/null &

    echo $! > "${prefix}.pid"
    printf 'started: %s.log\n' "$prefix"
}

# SS源模型：评测 SS、MS、LS
eval_mahdrl_as_fast_hscore_all S Tight
eval_mahdrl_as_fast_hscore_all S Medium
eval_mahdrl_as_fast_hscore_all S Loose

# SM源模型：评测 SM、MM、LM
eval_mahdrl_as_fast_hscore_all M Tight
eval_mahdrl_as_fast_hscore_all M Medium
eval_mahdrl_as_fast_hscore_all M Loose

# SL源模型：评测 SL、ML、LL
eval_mahdrl_as_fast_hscore_all L Tight
eval_mahdrl_as_fast_hscore_all L Medium
eval_mahdrl_as_fast_hscore_all L Loose
