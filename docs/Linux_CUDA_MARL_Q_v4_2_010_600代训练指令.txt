# Derived from MARL-Q4.2; delay=0.1, energy=0.9; idle VM only; SR>=92%.
# MARL-Q marl_q_v4_2_010: 600 episodes, nine source training scenarios, SR >= 92% selection.
# Run from the project root in Bash with the Python environment activated.

train_marl_v4_2_010() {
    local res="$1" ddl="$2"
    local prefix="outputs/server_console_logs/marl_q_v4_2_010_S${res}_${ddl}_m1"
    mkdir -p outputs/server_console_logs || return 1
    CUDA_VISIBLE_DEVICES=0 python -c \
        'import torch; assert torch.cuda.is_available(), "CUDA unavailable"; torch.ones(1, device="cuda")' || return 1
    CUDA_VISIBLE_DEVICES=0 nohup python -u launch.py algorithms/marl_q_v4_2_010/train.py \
        --task S --resource "$res" --deadline "$ddl" \
        --episodes 600 --model-seed 1 \
        --experiment-protocol configs/f22_split_v1.json \
        --normalization-mode f19_shared_v1 \
        > "${prefix}.log" 2>&1 &
    echo $! > "${prefix}.pid"
}

# SS: small tasks, small resources
train_marl_v4_2_010 S Tight
train_marl_v4_2_010 S Medium
train_marl_v4_2_010 S Loose

# SM: small tasks, medium resources
train_marl_v4_2_010 M Tight

train_marl_v4_2_010 M Loose
