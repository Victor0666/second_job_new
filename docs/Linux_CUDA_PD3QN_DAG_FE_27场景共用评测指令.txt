# Linux / Bash：PD3QN-DAG-FE 27场景共用评测
# 每条命令加载一个已训练源模型，依次评测task=S/M/L：原训练规模1个+泛化2个。
# 源训练task=S固定，resource和deadline在组内不变；各目标30个实例，9组共27场景810次实例评测。
# 主CSV名包含目标场景与种子，例如eval_pd3qn_dag_fe_taskS_resS_Loose_model1_seeds700-729.csv；另有summary_pd3qn_dag_fe.json，各组共享SML控制台日志/PID。
# 从项目根目录、激活的Python/CUDA环境运行。先完成对应源训练，再逐组复制启动。
# CUDA_VISIBLE_DEVICES=0按实际GPU分配调整；--device cuda不可用会退出，不静默回退CPU。
# 任一目标失败时该组停止；非空结果目录拒绝覆盖，同名控制台日志重跑会覆盖。
# 如果原S目标已完成，将对应for task in S M L改成for task in M L，只补两个泛化目标。

mkdir -p outputs/server_eval_logs

# PD3QN-DAG-FE 评测 | 源SS/Tight -> SS、MS、LS / Tight
CUDA_VISIBLE_DEVICES=0 nohup bash -ec 'for task in S M L; do python -u launch.py algorithms/pd3qn_dag_fe/evaluate.py --task "$task" --resource S --deadline Tight --train-task S --train-resource S --train-deadline Tight --seeds 700-729 --model-seed 1 --experiment-protocol configs/f22_split_v1.json --normalization-mode pd3qn_units_v1 --device cuda; done' > outputs/server_eval_logs/eval_pd3qn_dag_fe_sourceSS_Tight_SML_select_sr099_esp099_v1_model1_seeds700-729.log 2>&1 &
echo $! > outputs/server_eval_logs/eval_pd3qn_dag_fe_sourceSS_Tight_SML_select_sr099_esp099_v1_model1_seeds700-729.pid
